cmake_minimum_required(VERSION 3.18)
project(libwsc_backpressure_tests LANGUAGES C CXX)
option(USE_TLS "Build TLS coverage" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(../libs/libwsc libwsc)
find_package(Threads REQUIRED)
add_executable(backpressure_client backpressure_client.cpp)
target_link_libraries(backpressure_client PRIVATE libwsc Threads::Threads)
target_compile_features(backpressure_client PRIVATE cxx_std_11)
add_executable(receiver_limits receiver_limits.cpp)
target_link_libraries(receiver_limits PRIVATE libwsc Threads::Threads)
